#!/bin/bash

# Obtener el directorio actual del script y definir la ruta por defecto del objetivo (Reporte_Diario.sh)
DIR_ACTUAL="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
SCRIPT_OBJETIVO="$DIR_ACTUAL/Reporte_Diario.sh"
ARCHIVO_SALIDA="$DIR_ACTUAL/Informe_Tareas.txt"


echo ""
echo "============================================="
echo "    GESTOR DE TAREAS AUTOMÁTICAS (CRON)      "
echo "============================================="
echo "1) Mostrar tareas programadas"
echo "2) Programar una nueva tarea"
echo "3) Eliminar tarea existente"
echo ""

#Pedimos la opc a elegir
read -p "Elija una opción (1-3): " opcion_menu

case $opcion_menu in
    1)
        echo ""
        echo "--- Tareas programadas actuales ---"
        if crontab -l 2>/dev/null | grep -q "."; then
            crontab -l
        else
            echo "No hay tareas programadas en este momento."
        fi
        echo ""
        ;;
    2)
        echo ""
        echo "--- Programar Nueva Tarea ---"

        # Bucle para solicitar y validar la hora (0-23)
        while true; do
            read -p "Ingrese la hora de ejecución (0-23): " hora
            if [[ "$hora" =~ ^[0-9]+$ ]] && [ "$hora" -ge 0 ] && [ "$hora" -le 23 ]; then
                break
            else
                echo "Error: Hora inválida. Debe ser un número entre 0 y 23."
                echo ""
            fi
        done

        # Bucle para solicitar y validar el minuto (0-59)
        while true; do
            read -p "Ingrese el minuto de ejecución (0-59): " minuto
            if [[ "$minuto" =~ ^[0-9]+$ ]] && [ "$minuto" -ge 0 ] && [ "$minuto" -le 59 ]; then
                break
            else
                echo "Error: Minuto inválido. Debe ser un número entre 0 y 59."
                echo ""
            fi
        done

        echo ""
        echo "Seleccione la frecuencia de la tarea:"
        echo "1. Una sola vez (hoy)"
        echo "2. Diaria"
        echo ""

        # Bucle para validar la frecuencia elegida
        while true; do
            read -p "Elija una opción (1-2): " freq_opcion
            if [ "$freq_opcion" = "1" ]; then
                dia_actual=$(date +%d)
                mes_actual=$(date +%m)
                expresion_cron="$minuto $hora $dia_actual $mes_actual *"
                break
            elif [ "$freq_opcion" = "2" ]; then
                expresion_cron="$minuto $hora * * *"
                break
            else
                echo "Opción inválida. Ingrese 1 o 2."
                echo ""
            fi
        done

        # Construir el comando para cron: ejecuta el script y redirige la salida acumulando en Informa_Tares.txt
        comando_cron="$expresion_cron /bin/bash '$SCRIPT_OBJETIVO' >> '$ARCHIVO_SALIDA' 2>&1"

        # Registrar en crontab sin borrar las tareas previas
        (crontab -l 2>/dev/null; echo "$comando_cron") | crontab -

        echo ""
        echo "¡Tarea programada exitosamente!"
        echo "Los resultados se guardarán automáticamente en: Informa_Tares.txt"
        ;;
    3)
        echo ""
        echo "--- Eliminar Tarea Programada ---"
        if ! crontab -l 2>/dev/null | grep -q "."; then
            echo "No hay tareas programadas para eliminar."
            echo ""
            exit 1
        fi

        echo "Tareas actuales:"
        mapfile -t tareas < <(crontab -l 2>/dev/null)
        echo ""

        # Mostrar de forma amigable (solo hora/minuto y nombre del script ejecutado)
        for i in "${!tareas[@]}"; do
            linea_cron="${tareas[$i]}"
            min_hr=$(echo "$linea_cron" | awk '{print $1, $2}')
            script_ejecutado=$(echo "$linea_cron" | grep -oE "/[^[:space:]]+\.sh" | tail -n 1)

            echo "$((i+1))) Hora programada [Min: ${min_hr%% *}, Hora: ${min_hr##* }] -> Archivo: ${script_ejecutado:-$linea_cron}"
        done

        # Bucle para reintentar si el número de tarea no es válido
        while true; do
            read -p "Ingrese el número de la tarea que desea eliminar: " num_tarea
            if [[ "$num_tarea" =~ ^[0-9]+$ ]] && [ "$num_tarea" -ge 1 ] && [ "$num_tarea" -le "${#tareas[@]}" ]; then
                indice_real=$((num_tarea - 1))

                {
                    for i in "${!tareas[@]}"; do
                        if [ "$i" -ne "$indice_real" ]; then
                            echo "${tareas[$i]}"
                        fi
                    done
                } | crontab -

                echo "Tarea eliminada exitosamente del sistema."
                echo ""
                break
            else
                echo "Número inválido. Elija un número de la lista mostrada."
                echo ""
            fi
        done
        ;;
    *)
        echo "Opción de menú inválida. Saliendo del programa..."
        echo ""
        exit 1
        ;;
esac
